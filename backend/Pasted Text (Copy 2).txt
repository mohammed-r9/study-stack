package main

import (
	"fmt"
	"net/http"
	"os"
	"os/exec"
	"path/filepath"
	"time"
)

func runAdminCommand(command string) error {
	cmd := exec.Command(
		"osascript",
		"-e",
		fmt.Sprintf(`do shell script "%s" with administrator privileges`, command),
	)

	cmd.Stdout = os.Stdout
	cmd.Stderr = os.Stderr

	return cmd.Run()
}

func runCommand(dir string, name string, args ...string) error {
	cmd := exec.Command(name, args...)
	cmd.Dir = dir
	cmd.Stdout = os.Stdout
	cmd.Stderr = os.Stderr
	cmd.Stdin = os.Stdin

	return cmd.Start()
}

func serveFrontend(distPath string) {
	fs := http.FileServer(http.Dir(distPath))

	fmt.Println("Serving frontend on http://localhost:3000")

	err := http.ListenAndServe(":3000", fs)
	if err != nil {
		panic(err)
	}
}

func openBrowser(url string) {
	exec.Command("open", url).Start()
}

func main() {

	root, err := os.Getwd()
	if err != nil {
		panic(err)
	}

	backend := filepath.Join(root, "backend")
	dist := filepath.Join(root, "dist")

	fmt.Println("Building Docker images...")

	err = runAdminCommand(fmt.Sprintf("cd %s && docker compose build", backend))
	if err != nil {
		panic(err)
	}

	fmt.Println("Starting backend server...")

	err = runCommand(backend, "./server")
	if err != nil {
		panic(err)
	}

	time.Sleep(2 * time.Second)

	go serveFrontend(dist)

	time.Sleep(1 * time.Second)

	fmt.Println("Opening browser...")

	openBrowser("http://localhost:3000")

	select {}
}
